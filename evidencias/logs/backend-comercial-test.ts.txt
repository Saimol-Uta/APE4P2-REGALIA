  it('CP-006: ADMIN external declared payment in CASH without simulated card debit or bank fields', async () => {
    const paymentAttemptsBefore = countPaymentAttempts();
    const cardBalanceBefore = getApprovedCardBalanceCents();

    const created = await http()
      .post('/api/v1/orders/admin')
      .set(bearer(adminToken))
      .set('Idempotency-Key', randomUUID())
      .send(
        baseAdminOrderPayload({

            method: 'CASH',
            note: 'Pago en caja',
          },
        }),
      )
      .expect(201);

    expect(created.body.payment).toEqual({
      kind: 'ADMIN_DECLARED_EXTERNAL',
      status: 'DECLARED',
      method: 'CASH',
      declaredById: adminId,
      declaredAt: expect.any(String),
    });
    expect(created.body.payment).not.toHaveProperty('cardLast4');
    expect(created.body.payment).not.toHaveProperty('reference');

    const receipt = await http()
      .get(`/api/v1/orders/${orderId}/receipt`)
      .set(bearer(adminToken))
      .expect(200);
    expect(receipt.body.payment).toEqual({
      kind: 'ADMIN_DECLARED_EXTERNAL',
      status: 'DECLARED',
      method: 'CASH',
      declaredById: adminId,
      declaredAt: created.body.payment.declaredAt,
    });
    expect(receipt.body.payment).not.toHaveProperty('cardLast4');
    expect(receipt.body.payment).not.toHaveProperty('reference');

    const declarationRow = sqlScalar(
      `SELECT method::text || '|' || "declaredById" || '|' || COALESCE(note, '') FROM public."externalPaymentDeclaration" WHERE "orderId"='${orderId}'`,
    );
    expect(declarationRow).toBe(`CASH|${adminId}|Pago en caja`);
    expect(countPaymentAttempts()).toBe(paymentAttemptsBefore);
    expect(getApprovedCardBalanceCents()).toBe(cardBalanceBefore);
  });

  it('CP-008: tax rate change after shop quote rejects checkout with 409 SHOP_SETTINGS_CHANGED and zero side effects', async () => {
    await ensureTaxRateBps(1500);
    const settingsBefore = await http().get('/api/v1/admin/shop-settings').set(bearer(adminToken)).expect(200);
    const ordersBefore = countOrders();
    const receiptsBefore = countReceipts();
    const stockBefore = getProductStock(productId);
    const balanceBefore = getApprovedCardBalanceCents();

    const quoteRes = await http()
      .post('/api/v1/shop/checkout/quote')
      .set(bearer(clientAToken))
      .send({
        items: [{ productId, quantity: 1 }],
        delivery: baseShopDelivery(),
      })
      .expect(200);
    expect(quoteRes.body.tax.rateBps).toBe(1500);

    await http()
      .patch('/api/v1/admin/shop-settings')
      .set(bearer(adminToken))
      .send({
        expectedVersion: settingsBefore.body.version,
        tax: { rateBps: 1200 },
      })
      .expect(200);

    const checkoutRes = await http()
      .post('/api/v1/shop/checkout')
      .set(bearer(clientAToken))
      .set('Idempotency-Key', randomUUID())
      .send({
        items: [{ productId, quantity: 1 }],
        delivery: baseShopDelivery(),
        paymentCard: approvedCard,
        quoteToken: quoteRes.body.quoteToken,
      })
      .expect(409);

    expect(checkoutRes.body.error.code).toBe('SHOP_SETTINGS_CHANGED');
    expect(countOrders()).toBe(ordersBefore);
    expect(countReceipts()).toBe(receiptsBefore);
    expect(getProductStock(productId)).toBe(stockBefore);
    expect(getApprovedCardBalanceCents()).toBe(balanceBefore);
  });
