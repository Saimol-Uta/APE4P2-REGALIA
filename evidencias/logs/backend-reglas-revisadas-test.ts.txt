function localDay(offsetDays: number): string {
  return Temporal.Now.zonedDateTimeISO(BUSINESS_TZ)
    .toPlainDate()
    .add({ days: offsetDays })
    .toString();
}

function localOffsetIso(offsetDays: number, plainTime: string): string {
  const dateStr = localDay(offsetDays);
  return `${dateStr}T${plainTime}-05:00`;
}

function localInstant(offsetDays: number, plainTime: string): Temporal.Instant {
  return Temporal.Now.zonedDateTimeISO(BUSINESS_TZ)
    .toPlainDate()
    .add({ days: offsetDays })
    .toZonedDateTime({ timeZone: BUSINESS_TZ, plainTime })
    .toInstant();
}

function formatGuayaquilClock(isoInstant: string): string {
  const zdt = Temporal.Instant.from(isoInstant).toZonedDateTimeISO(BUSINESS_TZ);
  const hh = String(zdt.hour).padStart(2, '0');
  const mm = String(zdt.minute).padStart(2, '0');
  return `${hh}:${mm}`;
}

interface CustomOsrmServer {
  url: string;
  setTableOverride: (fn: ((rawUrl: string) => { status: number; body: unknown } | null) | null) => void;
  close: () => Promise<void>;
}

async function startCaseOsrmServer(): Promise<CustomOsrmServer> {
  let tableOverride: ((rawUrl: string) => { status: number; body: unknown } | null) | null = null;
  const server: Server = createServer((req, res) => {
    const rawUrl = req.url ?? '/';
    const overridden = tableOverride ? tableOverride(rawUrl) : null;
    const { status, body } = overridden ?? fakeOsrmResponse(rawUrl, { mode: 'ok' });
    res.writeHead(status, {
      'content-type': typeof body === 'string' ? 'text/html' : 'application/json',
    });
    res.end(typeof body === 'string' ? body : JSON.stringify(body));
  });
  await new Promise<void>((resolve) => server.listen(0, '127.0.0.1', resolve));
  const { port } = server.address() as AddressInfo;
  return {
    url: `http://127.0.0.1:${port}`,
    setTableOverride: (fn) => {
      tableOverride = fn;
    },
    close: () =>
      new Promise<void>((resolve) => {
        server.close(() => resolve());
        server.closeAllConnections?.();
      }),
  };
}

interface CaseHarness {
  app: INestApplication;
  db: { client: ReturnType<typeof postgres<Contract>>; user: any };
  pgClient: ReturnType<typeof postgres<Contract>>;
  adminId: string;
  clientId: string;
  driverUserId: string;
  driverId: string;
  zoneId: string;
  categoryId: string;
  productId: string;
  adminToken: string;
  clientToken: string;
  driverToken: string;
}

async function closeCaseHarness(h: CaseHarness | undefined): Promise<void> {
  if (!h) return;
  const server = h.app.getHttpServer() as
    | {
        closeIdleConnections?: () => void;
        closeAllConnections?: () => void;
      }
    | undefined;
  try {
    server?.closeIdleConnections?.();
    await h.app.close();
  } finally {
    server?.closeAllConnections?.();
    await h.pgClient.close();
  }
}

      const today = localDay(0);
      const o1 = await seedPendingOrder(CLUSTER_POINT_A, localInstant(0, '10:00:00'), 'Stop 10:00');
      const o2 = await seedPendingOrder(CLUSTER_POINT_B, localInstant(0, '10:25:00'), 'Stop 10:25');

        .patch('/api/v1/admin/shop-settings')

        .expect(200);

      expect(groupedRes.body.distanceMethod).toBe('OSRM_ROAD_NETWORK');
      expect(groupedRes.body.roadMetrics).toMatchObject({ provider: 'OSRM', distanceUnit: 'km', durationUnit: 's' });
      expect(groupedRes.body.proposals).toHaveLength(1);
      expect(groupedRes.body.proposals[0].orderIds).toEqual([o1.id, o2.id, o3.id]);
      expect(groupedRes.body.proposals[0].totalDistanceKm).toBeGreaterThan(0);
      expect(groupedRes.body.proposals[0].totalDurationSeconds).toBeGreaterThan(0);
      expect(
        groupedRes.body.proposals[0].stops.every((s: any) => s.legDistanceKm > 0 && s.legDurationSeconds > 0),
      ).toBe(true);

      // Verify OSRM durationsS primary and distancesM secondary tie-breaking via GET /route-proposals
      const tieDay = localDay(1);
      const tieA = await seedPendingOrder({ lat: -1.235, lng: -78.613 }, localInstant(1, '10:00:00'), 'Tie A');
      const tieB = await seedPendingOrder({ lat: -1.239, lng: -78.609 }, localInstant(1, '10:00:00'), 'Tie B');

      // Matrix indices correspond to uniquePoints([depot, tieA, tieB]): 0=depot, 1=tieA, 2=tieB
      const buildTableBody = (durations: number[][], distances: number[][]) => {
        const coords = [
          [STORE_ORIGIN.lng, STORE_ORIGIN.lat],
          [-78.613, -1.235],
          [-78.609, -1.239],
        ];
        const wp = (i: number) => ({ location: coords[i], distance: 5, name: `W${i}` });
        return {
          code: 'Ok',
          durations,
          distances,
          sources: [0, 1, 2].map(wp),
          destinations: [0, 1, 2].map(wp),
        };
      };

      // Case 1: depot->A->B->depot has shorter duration (100+50+100=250s) than depot->B->A->depot (200+500+200=900s)
      // even though depot->A->B->depot has longer distance (5000+1000+5000=11000m > 1000+1000+1000=3000m).
      osrm.setTableOverride((rawUrl) => {
        if (!rawUrl.startsWith('/table/v1/')) return null;
        return {
          status: 200,
          body: buildTableBody(
            [
              [0, 100, 200],
              [200, 0, 50],
              [100, 500, 0],
            ],
            [
              [0, 5000, 1000],
              [1000, 0, 1000],
              [5000, 1000, 0],
            ],
          ),
        };
      });

      const durWin = await http()
        .get(`/api/v1/route-proposals?date=${tieDay}`)
        .set(auth(h.adminToken))
        .expect(200);
      expect(durWin.body.proposals).toHaveLength(1);
      expect(durWin.body.proposals[0].orderIds).toEqual([tieA.id, tieB.id]);

      // Case 2: Equal duration for both directions (100+50+100=250s vs 100+50+100=250s);
      // distancesM breaks the tie in favor of depot->B->A->depot (1000+1000+1000=3000m < 5000+1000+5000=11000m).
      osrm.setTableOverride((rawUrl) => {
        if (!rawUrl.startsWith('/table/v1/')) return null;
        return {
          status: 200,
          body: buildTableBody(
            [
              [0, 100, 100],
              [100, 0, 50],
              [100, 50, 0],
            ],
            [
              [0, 5000, 1000],
              [1000, 0, 1000],
              [5000, 1000, 0],
            ],
          ),
        };
      });

      const distTieWin = await http()
        .get(`/api/v1/route-proposals?date=${tieDay}`)
        .set(auth(h.adminToken))
        .expect(200);
      expect(distTieWin.body.proposals).toHaveLength(1);
      expect(distTieWin.body.proposals[0].orderIds).toEqual([tieB.id, tieA.id]);
    });

    it('CP-027: Liberación auditada de DRAFT caducado (RELEASE_DRAFT) y reprogramación HTTP (CONTACT + RESCHEDULE)', async () => {
      const yesterdayNoon = localInstant(-1, '12:00:00');
      const today = localDay(0);
      // Use the next business day in America/Guayaquil to avoid same-day clock boundary issues.
      const nextBusinessDay = localDay(1);
      const rescheduledDate = localInstant(1, '10:00:00');
      const rescheduledWindowStart = localInstant(1, '09:30:00');
      const rescheduledWindowEnd = localInstant(1, '10:30:00');

      const order = await seedPendingOrder(VALID_DELIVERY_POINT, yesterdayNoon, 'Draft Caducado St');
      const expiredRoute = await h.db.client.transaction(async (tx: any) => {
        const created = await tx.orm.public.Route.create({
          zoneId: h.zoneId,
          date: yesterdayNoon,
          status: 'DRAFT',
        });
        await tx.orm.public.OrderClaim.create({
          orderId: order.id,
          routeId: created.id,
          claimedAt: yesterdayNoon,
        });
        await tx.orm.public.RouteStop.create({
          routeId: created.id,
          orderId: order.id,
          sequence: 1,
          outcome: 'PENDING',
          isLegacy: false,
        });
        return created;
      });

      const assignRes = await http()
        .post(`/api/v1/routes/${expiredRoute.id}/assign`)
        .set(auth(h.adminToken))
        .send({ driverId: h.driverId })
        .expect(409);
      expect(assignRes.body.code).toBe('ROUTE_DATE_NOT_TODAY');

      const requestId = randomUUID();
      const interventionPayload = {
        action: 'RELEASE_DRAFT',
        routeId: expiredRoute.id,
        requestId,
        reason: 'Draft caducado del día operativo anterior',
      };

      const releaseRes = await http()
        .post('/api/v1/dispatch/interventions')
        .set(auth(h.adminToken))
        .send(interventionPayload)
        .expect(200);
      expect(releaseRes.body).toMatchObject({
        action: 'RELEASE_DRAFT',
        routeId: expiredRoute.id,
        requestId,
        outcome: 'APPLIED',
      });

      const stopsAfter = await h.db.client.orm.public.RouteStop.where({
        routeId: expiredRoute.id,
      }).all();
      expect(stopsAfter).toHaveLength(0);

      const claimsAfter = await h.db.client.orm.public.OrderClaim.where({
        routeId: expiredRoute.id,
      }).all();
      expect(claimsAfter).toHaveLength(1);
      expect(claimsAfter[0].releasedAt).not.toBeNull();

      const orderAfter = await h.db.client.orm.public.Order.where({ id: order.id }).first();
      expect(orderAfter.status).toBe('PENDING');
      expect(orderAfter.scheduledDeliveryDate).not.toBeNull();
      expect(
        Temporal.Instant.compare(
          Temporal.Instant.from(orderAfter.scheduledDeliveryDate!),

      const auditsAfter = await h.db.client.orm.public.DispatchIntervention.where({
        routeId: expiredRoute.id,
      }).all();
      expect(auditsAfter).toHaveLength(1);
      expect(auditsAfter[0].action).toBe('RELEASE_DRAFT');

      // Replay with identical requestId must be idempotent
      const replayRes = await http()
        .post('/api/v1/dispatch/interventions')
        .set(auth(h.adminToken))
        .send(interventionPayload)
        .expect(200);
      expect(replayRes.body.id).toBe(releaseRes.body.id);
      expect(
        await h.db.client.orm.public.DispatchIntervention.where({
          routeId: expiredRoute.id,
        }).all(),
      ).toHaveLength(1);

      // Original date remains yesterday, so today's proposals must not include the released order.
